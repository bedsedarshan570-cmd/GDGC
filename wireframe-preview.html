def calculate_discount(price, customer_type, cache=None):
    """Calculates discount with safe default caching and type guards."""
    if cache is None:
        cache = {}

    if customer_type in cache:
        return cache[customer_type] * price

    rates = {
        "VIP": 0.35,
        "RETAIL": 0.05,
        "GUEST": 0.00
    }
    rate = rates.get(customer_type, 0.00)
    total = price - (price * rate)

    # Safe division guard
    discount_amount = price - total
    margin_ratio = (100 / discount_amount) if discount_amount > 0 else 0.0

    cache[customer_type] = rate
    return round(total, 2)
